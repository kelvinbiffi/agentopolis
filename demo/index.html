Agentopolis

A tiny 3D city where your AI agents live. The one speaking walks to the crossroads and the agent it answers meets it there; quiet agents go for a coffee. Feed it events from any multi-agent system.

import { createAgentopolis } from '@kelvinbiffi/agentopolis'

const city = createAgentopolis(document.getElementById('city'), {
  agents: [
    { id: 'director', name: 'Director', color: '#f59e0b', sign: 'DIRECTOR' },
    { id: 'designer', name: 'Designer', color: '#8b5cf6', sign: 'ART DEPT' },
    { id: 'critic',   name: 'Critic',   color: '#10b981', sign: 'SCREENING' },
  ],
})

city.say('designer', 'Scene 1 is ready.')
city.say('critic', 'Scene 1: 6/10. More drama, please.')
city.done()
